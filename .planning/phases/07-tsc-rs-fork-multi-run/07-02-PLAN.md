**Planner decisions encoded here:**
1. **Reuse proof mechanism** — the `held` response block is the decisive proof (same
   `instanceGen` across two HTTP requests + `runsOnInstance` climbing); the timing drop
   (`results[1].ms < results[0].ms`) is asserted alongside as the performance corroboration.
   Timing alone cannot distinguish reuse from a fast second cold run.
2. **Fresh-vs-held bench mechanism** — `TSCHECKER_WASM_RECYCLE_AFTER:1` reuses the ENTIRE
   held-path machinery to produce fresh-instance semantics (drop + re-instantiate per run from
   the same compiled Module = exactly Phase 6's per-check behavior, 07-01 kept `runTsc` for
   provenance but nothing routes to it over HTTP). No second run path, no new endpoint.
3. **Memory smoke is a plausibility assert, not exactness** — bound `< 100 MB` of the 128 MB
   isolate (research plateau: 62,521,344 bytes) and `last <= run-3 sample + 2 MB` (research:
   52 KB growth across 50 runs). 07-CONTEXT's optional memory guard stays OUT of the shim
   (recycle-at-50 bounds it); this smoke is the observable check.
4. **The 20,000-deep guest** — the trigger is the host stack budget, not the exact depth; if a
   future workerd ever returns 200, deepen the repeat count rather than weakening the assert.

async function freePort() {
  return new Promise((resolve, reject) => {
    const srv = createServer();
    srv.listen(0, "127.0.0.1", () => {
      const chosen = srv.address().port;
      srv.close(() => resolve(chosen));
    });
    srv.on("error", reject);
  });
}

async function get(port, path, { method = "GET", body, headers } = {}) {
  const res = await fetch(`http://127.0.0.1:${port}${path}`, {
    method,
    headers,
    body,
    signal: AbortSignal.timeout(30000), // cold wasm checks cost ~0.5-1.2 s (07-RESEARCH)
  });
  let json = null;
  try {
    json = await res.json();
  } catch {
    json = null; // non-JSON body
  }
  return { status: res.status, headers: res.headers, json };
}

async function waitHealthy(port, getLog) {
  for (let i = 0; i < 60; i++) {
    await new Promise((r) => setTimeout(r, 1000));
    try {
      const res = await fetch(`http://127.0.0.1:${port}/healthz`, {

      if (res.ok) return { status: res.status, json: await res.json() };
    } catch {
      // not up yet
    }
  }
  throw new Error(`wrangler dev did not serve /healthz within 60s. Log tail:\n${getLog().slice(-2000)}`);
}

function spawnWrangler(port, vars) {
  const varFlags = Object.entries(vars).flatMap(([key, value]) => ["--var", `${key}:${value}`]);
  return spawn("wrangler", ["dev", "--port", String(port), ...varFlags], {
    env: { ...process.env, CI: "true" },
    stdio: ["ignore", "pipe", "pipe"],
    detached: true, // own process group so teardown kills wrangler + workerd together
  });
}

async function killWrangler(wrangler, port) {
  if (wrangler !== undefined && wrangler.pid !== undefined) {
    try {
      process.kill(-wrangler.pid, "SIGTERM");
    } catch {
      // already gone
    }
  }
  if (port !== undefined) {
    await new Promise((r) => setTimeout(r, 2000));
    let stillUp = true;
    try {
      await fetch(`http://127.0.0.1:${port}/healthz`, { signal: AbortSignal.timeout(1500) });
    } catch {
      stillUp = false; // down — good
    }
    if (stillUp) {
      try {
        execSync(`fuser -k ${port}/tcp 2>/dev/null || true`); // backstop; fuser may be absent
      } catch {
        // best effort
      }
      assert(false, "teardown: wrangler killed and port closed");
    } else {
      console.log("ok - teardown: wrangler killed and port closed");
    }
  }
}
