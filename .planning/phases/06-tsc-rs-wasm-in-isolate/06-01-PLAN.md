````markdown
## How the worker hosts it (WASM-01/02/03)

- **Import**: `import wasmModule from "./ts_rust.wasm"` — wrangler's default CompiledWasm rule
  (NO `rules` entry needed; same mechanism as `src/sandbox/quickjs.wasm`, which coexists —
  `npm run bundle:check` ships both). The import is a precompiled `WebAssembly.Module`; the
  only legal instantiation in workerd is with host-supplied imports, which the shim provides.
- **Shim**: `src/typecheck-wasm/shim.ts` is a workerd-safe TypeScript port of upstream
  `npm/wasm/core.js`: `ts_host.fs`/`fs_take` over `memfs.ts`'s in-memory map, a working WASI
  subset (`args_sizes_get`/`args_get`/`environ_sizes_get`/`environ_get`/`clock_time_get`/
  `clock_res_get`/`random_get`/`fd_write`/`fd_fdstat_get`/`fd_prestat_get`/`proc_exit`/
  `sched_yield`/`poll_oneoff`), and ENOSYS (52) stubs wired to EVERY other import the module
  actually declares (imports-driven loop, like core.js). NO Node APIs: core.js's Node-only bits
  (`worker_threads` 256 MB stacks, real fs) are deliberately not ported — the run is synchronous
  on the calling thread.
- **Fresh instance per check**: the Module is compiled once at bundle load; every check builds
  a new `WebAssembly.Instance` (2–8 ms measured — negligible next to the check). This matches
  the module's own one-run-per-instance design; no instance pooling.
- **Run protocol**: request = `[cwd, flags, ...args].join("\0")` UTF-8 written via
  `ts_input(len)`; `ts_run()` executes; diagnostics JSON comes back from
  `ts_output()`/`ts_output_len()`; `proc_exit` surfaces as a thrown `WasiExit` (normal-exit
  control flow, not a crash).
- **Libs are INSIDE the module** (embedded LZMA, decoded in-Rust) — no lib assets are vendored
  or seeded by the host. Guest files + a strict/es2022 tsconfig are seeded under `/app` per
  check; diagnostics fileNames are mapped back to the repo's rooted form by `mapping.ts`.
- **Flags used**: bit 0 (diagnostics JSON) only. `--watch`/`--lsp`/`--api` do not exist in the
  wasm build.

## Known limits (documented, not solved — 06-CONTEXT)

- **Stack depth**: the module links a 32 MiB in-linear-memory shadow stack
  (`-zstack-size=33554432`). With no JSPI in workerd, deep checker recursion can trap the
  instance; a trap surfaces as a thrown host error with captured stderr (same class as QuickJS
  aborts) and the route returns the structured 500 envelope. Guests in this prototype are small.
- **Sync CPU-bound run**: spike-measured per request in workerd — cold ≈ 1.22 s, warm median
  ≈ 625 ms, instantiate 2–8 ms (fresh instance each time). Local `wrangler dev` does not enforce
  CPU limits; production would need a paid plan's 30 s ceiling. The prototype is local-only.
- **Per-run instance cost** is a rounding error; no pooling (see above).
````

```ts
// In-memory filesystem behind ts_rust.wasm's ts_host.fs bridge (WASM-01): a
// TypeScript port of pingdotgg/ts-rust@569df52 npm/wasm/core.js's
// memoryFileSystem. PURE module — zero imports — so vitest specs exercise it
// directly instead of dragging the 5.1 MB wasm module into the worker
// transport (Phase 2 lesson). The checker seeds a fresh map per check; any
// writes from the module land in that map and die with the instance.
export interface MemFsStat {
  isDirectory: boolean;
  isFile?: boolean;
  size: number;
  mtimeMs?: number;
  mtimeNs?: bigint;
}

export interface MemFsEntry {
  name: string;
  kind: "file" | "directory" | "symlink" | "other";
}

export interface MemoryFileSystem {
  files: Map<string, string>;
  readFile(path: string): string | undefined;
  stat(path: string): MemFsStat | undefined;
  readDirectory(path: string): MemFsEntry[] | undefined;
  realpath(path: string): string | undefined;
  // Error string on failure, undefined on success (the shim's writeResult
  // maps both onto the fs-bridge protocol).
  writeFile(path: string, data: Uint8Array, append: boolean): string | undefined;
  remove(path: string): string | undefined;
  // Optional: the check filesystem never implements it; absence is a
  // success no-op (core.js contract — a read-only check never chtimes).
  chtimes?(path: string, atimeNs?: bigint, mtimeNs?: bigint): string | undefined;
}

const ENCODER = new TextEncoder();
const FILE_DECODER = new TextDecoder("utf-8", { ignoreBOM: true });

export function createMemoryFileSystem(files: Map<string, string> = new Map()): MemoryFileSystem {
  const map = files instanceof Map ? files : new Map(Object.entries(files));
  const dirPrefix = (path: string): string => (path.endsWith("/") ? path : `${path}/`);
  const isDir = (path: string): boolean => {
    if (path === "/") return true;
    const prefix = dirPrefix(path);
    for (const key of map.keys()) {
      if (key.startsWith(prefix)) return true;
    }
    return false;
  };
  return {
    files: map,
    readFile: (path) => map.get(path),
    stat(path) {
      const text = map.get(path);
      if (text !== undefined) {
        return { isDirectory: false, size: ENCODER.encode(text).length, mtimeMs: 0 };
      }
      return isDir(path) ? { isDirectory: true, size: 0, mtimeMs: 0 } : undefined;
    },
    readDirectory(path) {
      const prefix = dirPrefix(path);
      const entries = new Map<string, MemFsEntry["kind"]>();
      for (const key of map.keys()) {
        if (!key.startsWith(prefix)) continue;
        const rest = key.slice(prefix.length);
        const slash = rest.indexOf("/");
        const name = slash < 0 ? rest : rest.slice(0, slash);
        if (!entries.has(name)) entries.set(name, slash < 0 ? "file" : "directory");
      }
      return entries.size || isDir(path)
        ? [...entries].map(([name, kind]) => ({ name, kind }))
        : undefined;
    },
    realpath: (path) => (map.has(path) || isDir(path) ? path : undefined),
    writeFile(path, data, append) {
      if (isDir(path)) return `open ${path}: is a directory`;
      for (let slash = path.indexOf("/"); slash >= 0; slash = path.indexOf("/", slash + 1)) {
        const parent = path.slice(0, slash);
        if (map.has(parent)) return `mkdir ${parent}: not a directory`;
      }
      const text = FILE_DECODER.decode(data);
      map.set(path, append ? (map.get(path) ?? "") + text : text);
      return undefined;
    },
    remove(path) {
      const prefix = dirPrefix(path);
      for (const key of [...map.keys()]) {
        if (key === path || key.startsWith(prefix)) map.delete(key);
      }
      return undefined;
    },
  };
}
```

```ts
// Pure mapping from ts_goport's JSON diagnostics shape to the repo's flat
// TypecheckDiagnostic. ZERO runtime imports (the only import is a type,
// erased at emit), so vitest specs exercise it without dragging the 5.1 MB
// wasm module OR the 8.8 MB typescript build into the worker transport
// (Phase 2 lesson).
import type { TypecheckDiagnostic } from "../typecheck/index";

// Wire shape (06-RESEARCH probe transcripts): file diagnostics carry
// startPosition {line, character} ZERO-based plus sourceLines/endPosition;
// config diagnostics carry pos/end = -1 and NO startPosition/fileName.
// `category` is tsc's numeric enum (0 Warning, 1 Error, 2 Message,
// 3 Suggestion).
export interface WasmSourcePosition {
  line: number;
  character: number;
}

export interface WasmDiagnostic {
  fileName?: string;
  startPosition?: WasmSourcePosition;
  code: number;
  category: number;
  text: string;
}

const CATEGORY_NAMES = ["Warning", "Error", "Message", "Suggestion"] as const;

// Guests are seeded under /app (the spike-proven tsconfig layout — see
// checker.ts); reported fileNames are stripped back to the repo's rooted
// form ("/app/guest.ts" -> "/guest.ts") so diagnostics are byte-identical
// across checker implementations.
function repoFileName(fileName: string): string {
  return fileName.startsWith("/app/") ? `/${fileName.slice("/app/".length)}` : fileName;
}

export function mapWasmDiagnostic(raw: WasmDiagnostic): TypecheckDiagnostic {
  const start = raw.startPosition;
  return {
    code: raw.code,
    message: raw.text,
    line: start ? start.line + 1 : -1,
    column: start ? start.character + 1 : -1,
    file: raw.fileName !== undefined ? repoFileName(raw.fileName) : "(program)",
    category: CATEGORY_NAMES[raw.category] ?? String(raw.category),
  };
}

// Same ordering semantics as the ts-js checker's byLocation (file, line,
// column) — duplicated here because importing it would drag the typescript
// runtime into this module's graph.
function byLocation(a: TypecheckDiagnostic, b: TypecheckDiagnostic): number {
  return a.file.localeCompare(b.file) || a.line - b.line || a.column - b.column;
}

export function mapWasmDiagnostics(raw: WasmDiagnostic[]): TypecheckDiagnostic[] {
  return raw.map(mapWasmDiagnostic).sort(byLocation);
}
```

```ts
// fs-bridge results: staged bytes, or an error string the module surfaces
// through the -2-len return convention.
type FsCallResult = { bytes: Uint8Array } | { error: string } | undefined;

function writeResult(result: string | undefined): FsCallResult {
  return typeof result === "string" ? { error: result } : { bytes: new Uint8Array() };
}

// Decode one ts_host.fs op against the check's in-memory filesystem.
// Request bytes at ptr/len are UTF-8, NUL-separated for write/chtimes.
function fsCall(fs: MemoryFileSystem, op: number, request: Uint8Array): FsCallResult {
  const text = DECODER.decode(request);
  switch (op) {
    case OP_READ: {
      const data = fs.readFile(text);
      return data === undefined ? undefined : { bytes: ENCODER.encode(data) };
    }
    case OP_STAT: {
      const stat = fs.stat(text);
      if (!stat) return undefined;
      const kind = stat.isDirectory ? "d" : stat.isFile === false ? "o" : "f";
      const ns = stat.mtimeNs ?? BigInt(Math.round((stat.mtimeMs ?? 0) * 1e6));
      return { bytes: ENCODER.encode(`${kind} ${stat.size ?? 0} ${ns}`) };
    }
    case OP_READ_DIR: {
      const entries = fs.readDirectory(text);
      if (!entries) return undefined;
      return {
        bytes: ENCODER.encode(entries.map((e) => `${KIND_CHAR[e.kind] ?? "o"}${e.name}\0`).join("")),
      };
    }
    case OP_REALPATH: {
      const real = fs.realpath(text);
      return real === undefined ? undefined : { bytes: ENCODER.encode(real) };
    }
    case OP_WRITE:
    case OP_APPEND: {
      const nul = request.indexOf(0);
      const path = DECODER.decode(request.subarray(0, nul));
      const data = request.subarray(nul + 1);
      return writeResult(fs.writeFile(path, data, op === OP_APPEND));
    }
    case OP_REMOVE:
      return writeResult(fs.remove(text));
    case OP_CHTIMES: {
      const [path, atime, mtime] = text.split("\0");
      if (!fs.chtimes) return { bytes: new Uint8Array() };
      return writeResult(
        fs.chtimes(path, atime ? BigInt(atime) : undefined, mtime ? BigInt(mtime) : undefined),
      );
    }
    default:
      return undefined;
  }
}

    const view = dv();

// WebAssembly.Instance per call (CONTEXT decision); the Module itself is
// compiled once at bundle load (CompiledWasm import).

  const stderrChunks: string[] = [];

    args: options.args,
    cwd: options.cwd,
    fs: createMemoryFileSystem(options.files),
    env: options.env,
    stdout: (chunk) => stdoutChunks.push(STREAM_DECODER.decode(chunk, { stream: true })),
    stderr: (chunk) => stderrChunks.push(STREAM_DECODER.decode(chunk, { stream: true })),
    diagnosticsJson: options.diagnosticsJson,
  });
  const t0 = performance.now();
  const instance = new WebAssembly.Instance(wasmModule, run.imports);
  const t1 = performance.now();
  const tsRun = run.start(instance);
  let exitCode = SUCCESS;
  let error: unknown;
  try {
    exitCode = tsRun();
  } catch (thrown) {
    error = thrown;

  return {
    exitCode: code,
    diagnostics,
    stdout: stdoutChunks.join("").slice(0, 65536),
    stderr: stderrChunks.join("").slice(0, 65536),
    timings: { instantiateMs: t1 - t0, runMs: t2 - t1 },
  };
}
```

```ts
// TscRsWasmTypechecker (WASM-02/03): the real Rust checker (ts_goport built
// to wasm32-wasip1) running INSIDE the workerd isolate behind the same
// Typechecker seam as TsTypechecker. One WebAssembly.Instance per check
// (the module's one-run-per-instance design; 2-8 ms instantiate — 06-
// RESEARCH), the libs are embedded in the module, and guest files ride the
// shim's in-memory fs bridge. Diagnostics arrive as ts_goport JSON and are
// mapped by ./mapping onto the exact flat shape the ts-js checker emits.
import { GUEST_GLOBALS_TEXT } from "../typecheck/vfs";
import type { TypecheckResult, Typechecker } from "../typecheck/index";
import { mapWasmDiagnostics } from "./mapping";
import { runTsc, TS_RUST_PIN } from "./shim";

// The spike-proven project config seeded into every check (06-RESEARCH
// probe requests): strict + es2022, wildcard include relative to /app.
// CLI args add --noEmit; everything else is the module's embedded default.
const TSCONFIG_TEXT = '{"compilerOptions":{"strict":true,"target":"es2022"},"include":["*.ts"]}';

export class TscRsWasmTypechecker implements Typechecker {
  readonly id = `ts-rust@${TS_RUST_PIN} (wasm)`;

  async check(files: Map<string, string>): Promise<TypecheckResult> {
    // Repo paths are rooted ("/guest.ts"); the module's project root is
    // /app (the tsconfig home) — guests are seeded one level down and
    // diagnostics are mapped back (mapping.ts) so reported files stay
    // byte-identical across checker implementations.
    const wasmFiles = new Map<string, string>([["/app/tsconfig.json", TSCONFIG_TEXT]]);
    for (const [name, content] of files) {
      wasmFiles.set(`/app${name.startsWith("/") ? name : `/${name}`}`, content);
    }
    // Ambient globals (test/expect/assert/console): include ["*.ts"] picks
    // the .d.ts up; without this seed every test() guest drowns in TS2593
    // (same rootName lesson as the ts-js checker and the 05-01 bench).
    wasmFiles.set("/app/guest-globals.d.ts", GUEST_GLOBALS_TEXT);
    const t0 = performance.now();
    const run = runTsc({
      files: wasmFiles,
      args: ["-p", "/app", "--noEmit"],
      cwd: "/",
      env: { NO_COLOR: "1" },
      diagnosticsJson: true,
    });
    const ms = performance.now() - t0;
    const diagnostics = mapWasmDiagnostics(run.diagnostics);
    return { ok: diagnostics.every((d) => d.category !== "Error"), diagnostics, ms };
  }
}
```

```ts
// TSCHECKER env selection (WASM-03): pure parsing, zero imports, so vitest
// specs exercise it without pulling in either checker graph (typescript
// 8.8 MB or the 5.1 MB wasm module — the Phase 2 transport lesson).
export type TSCheckerSelection = "ts-js" | "tsc-rs-wasm";

export function parseTSChecker(env?: { TSCHECKER?: string }): TSCheckerSelection {
  const raw = env?.TSCHECKER;

  if (raw === "tsc-rs-wasm") return "tsc-rs-wasm";

}
```
