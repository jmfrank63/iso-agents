# Plan 04-01: Agent loop core (wave 1)

## Objective

Build every pure/small-module piece of the Phase 4 agent loop so 04-02 only wires HTTP:
the `LlmProvider` seam + OpenAI-compatible provider + factory (LLM-01), the
deterministic broken-first mock (LLM-02), the robust JSON-contract extractor with the

vitest-testable under the small-module discipline: NO vitest spec may transitively pull

**Vitest-safety invariants this plan enforces mechanically:**

- `src/agent/loop.ts` imports the Typechecker/SandboxRuntime/TranspileResult/Llm seams

  graph and crash the vitest worker transport.
- `src/agent/provider.ts` imports `mock.ts` at runtime (factory); `mock.ts` imports
  provider types only (erased) — no runtime cycle, both vitest-safe.

## Discovered necessity (flagged before implementation)

`GUEST_GLOBALS_TEXT` (src/typecheck/vfs.ts:14) declares only `console`. Every mock
scenario — and every honest LLM answer following the system prompt — calls `test(...)` /
`expect(...)`. Under `strict` those are TS2304 `Cannot find name` ERRORS, so the
typecheck gate would fail EVERY iteration and the mock could never converge. T3 extends
the guest-globals declaration surface to mirror exactly what the Phase 3 sandbox
injects at guest runtime (src/sandbox/harness.ts): `test`, `assert`, `expect` with
`toEqual`/`toBeTrue`/`toBeCloseTo`. The Phase 2 harness cases (GREEN/RED/LIB_GUEST) use
none of these names and are unaffected; the 04-02 chain re-proves the Phase 2 harness
live after this change.

// LlmProvider seam (LLM-01/03): the agent loop depends on this interface,
// never on a vendor SDK. OpenAI-compatible = one raw fetch POST (STACK
// decision); the deterministic mock is the zero-key default. Error bodies
// are NEVER parsed (compat servers vary) — status + a short text snippet.

export interface LlmMessage {
  role: "system" | "user" | "assistant";
  content: string;
}

export interface LlmUsage {
  promptTokens: number;
  completionTokens: number;
}

export interface LlmCompletion {
  text: string;
  usage?: LlmUsage;
}

export class LlmError extends Error {
  constructor(
    message: string,
    readonly retryable: boolean,
    readonly detail?: unknown,
  ) {
    super(message);
  }
}

export interface LlmProvider {
  readonly id: string;
  complete(messages: LlmMessage[]): Promise<LlmCompletion>;
}

export interface LlmProviderEnv {
  LLM_PROVIDER?: string;
  LLM_BASE_URL?: string;
  LLM_API_KEY?: string;
  LLM_MODEL?: string;
  LLM_TIMEOUT_MS?: string;
}

export const OPENAI_PROVIDER_ID = "openai-compatible";

export const DEFAULT_LLM_TIMEOUT_MS = 20000;

const MAX_ERROR_BODY_CHARS = 500;

// "mock" (default/unset) -> MockLlmProvider; "openai" -> OpenAiCompatibleLlmProvider.
// Constructed PER REQUEST (the mock's call counter is per-run state — no
// cross-run races, research recommendation). "openai" without LLM_BASE_URL/
// LLM_MODEL throws here; the route catches it and finalizes the run as
// llm_error (honest misconfiguration surface).
export function createLlmProvider(env: LlmProviderEnv, timeoutMs: number): LlmProvider {
  const kind = env.LLM_PROVIDER ?? "mock";
  if (kind === "mock") {
    return new MockLlmProvider();
  }
  if (kind === "openai") {
    if (
      env.LLM_BASE_URL === undefined ||
      env.LLM_BASE_URL === "" ||
      env.LLM_MODEL === undefined ||
      env.LLM_MODEL === ""
    ) {
      throw new LlmError("LLM_PROVIDER=openai requires LLM_BASE_URL and LLM_MODEL", false);
    }
    return new OpenAiCompatibleLlmProvider(env.LLM_BASE_URL, env.LLM_MODEL, env.LLM_API_KEY, timeoutMs);
  }
  throw new LlmError(`unknown LLM_PROVIDER "${kind}" (use "mock" or "openai")`, false);
}

// stays stable). Only "openai" selects the openai identity; anything else —
// including a bogus value — reports the mock identity, while createLlmProvider
// is the one that rejects bogus values at request time.
export function getAgentIdentity(env: LlmProviderEnv): string {
  return env.LLM_PROVIDER === "openai" ? "openai-compatible" : "mock-provider ready";
}

// One chat-completions call with an internal retry-once for retryable
// failures (5xx/429/timeout/network — CONTEXT: "one retry on 5xx/timeout").
// TimeoutError (AbortSignal) and fetch refusals (TypeError) are classified
// by err.name, never instanceof (research Gotcha 7).
export class OpenAiCompatibleLlmProvider implements LlmProvider {
  readonly id = OPENAI_PROVIDER_ID;

  constructor(
    private readonly baseUrl: string,
    private readonly model: string,
    private readonly apiKey: string | undefined,
    private readonly timeoutMs: number,
  ) {}

  async complete(messages: LlmMessage[]): Promise<LlmCompletion> {
    try {
      return await this.callOnce(messages);
    } catch (err) {
      if (err instanceof LlmError && err.retryable) {
        return await this.callOnce(messages);
      }
      throw err;
    }
  }

  private async callOnce(messages: LlmMessage[]): Promise<LlmCompletion> {
    let res: Response;
    try {
      res = await fetch(`${this.baseUrl}/chat/completions`, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          ...(this.apiKey !== undefined && this.apiKey !== ""
            ? { authorization: `Bearer ${this.apiKey}` }
            : {}),
        },
        body: JSON.stringify({ model: this.model, messages, temperature: 0.2 }),
        signal: AbortSignal.timeout(this.timeoutMs),
      });
    } catch (err) {
      const name = err instanceof Error ? err.name : String(err);
      const detail = err instanceof Error ? err.message : String(err);
      throw new LlmError(`llm request failed (${name})`, true, detail);
    }
    if (!res.ok) {
      const body = (await res.text().catch(() => "")).slice(0, MAX_ERROR_BODY_CHARS);
      const retryable = res.status >= 500 || res.status === 429;
      throw new LlmError(`llm returned HTTP ${res.status}`, retryable, body);
    }
    const data = (await res.json().catch(() => null)) as ChatCompletionsResponse | null;
    const content = data?.choices?.[0]?.message?.content;
    if (typeof content !== "string") {
      throw new LlmError("llm response carried no assistant content", false);
    }
    const usage = data?.usage;
    return {
      text: content,
      ...(usage?.prompt_tokens !== undefined && usage?.completion_tokens !== undefined
        ? { usage: { promptTokens: usage.prompt_tokens, completionTokens: usage.completion_tokens } }
        : {}),
    };
  }
}

interface ChatCompletionsResponse {
  choices?: Array<{ message?: { content?: string | null } }>;
  usage?: { prompt_tokens?: number; completion_tokens?: number };
}

// Runtime import on purpose (factory above); mock.ts imports provider TYPES
// only, so this edge stays one-directional at runtime.
import { MockLlmProvider } from "./mock";
```

```ts
// Contract extraction (CONTEXT): the LLM must answer ONLY
// {"explanation": string, "files": Record<name, content>}. Real providers
// wrap that in markdown fences; strip them, then JSON.parse, then validate
// shape. Malformed -> MalformedContractError (the LOOP retries once with the
// corrective message, then terminates llm_malformed). import/export in any
// file -> ScriptViolationError (guests are script-style: transpileModule
// keeps `export`, which is a SyntaxError under QuickJS strict global eval —
// research-verified). "Malformed -> null" from the planning brief is
// realized as this typed error: a malformed answer NEVER yields a partial
// contract.
export interface LlmContract {
  explanation: string;
  files: Record<string, string>;
}

export class MalformedContractError extends Error {}

export class ScriptViolationError extends Error {
  constructor(readonly token: string) {
    super(`guest files must be script-style: source contains "${token}" (no import/export statements)`);
  }
}

// Same conservative trade as the sandbox's async gate: a word-boundary match
// in a comment is rejected too.
const SCRIPT_VIOLATION_PATTERN = /\b(?:import|export)\b/;

export function findScriptViolation(source: string): string | null {
  const match = SCRIPT_VIOLATION_PATTERN.exec(source);
  return match === null ? null : match[0];
}

export function extractContract(text: string): LlmContract {
  const stripped = stripFences(text);
  let parsed: unknown;
  try {
    parsed = JSON.parse(stripped);
  } catch {
    parsed = trySalvage(stripped);
  }
  if (parsed === null || typeof parsed !== "object" || Array.isArray(parsed)) {
    throw new MalformedContractError("contract is not a JSON object");
  }
  const obj = parsed as { explanation?: unknown; files?: unknown };
  if (obj.files === null || typeof obj.files !== "object" || Array.isArray(obj.files)) {
    throw new MalformedContractError("contract.files must be an object of {name: content}");
  }
  const files: Record<string, string> = {};
  for (const [name, content] of Object.entries(obj.files)) {
    if (typeof content !== "string") {
      throw new MalformedContractError(`contract.files.${name} must be a string`);
    }
    files[name] = content;
  }
  if (Object.keys(files).length === 0) {
    throw new MalformedContractError("contract.files must contain at least one file");
  }
  for (const [name, source] of Object.entries(files)) {
    const token = findScriptViolation(source);
    if (token !== null) {
      throw new ScriptViolationError(token);
    }
  }
  return { explanation: typeof obj.explanation === "string" ? obj.explanation : "", files };
}

// ```json ... ``` / ``` ... ``` — only a LEADING fence is stripped here; a
// missing trailing fence still parses via the brace-salvage path.
function stripFences(text: string): string {
  const trimmed = text.trim();
  const open = trimmed.match(/^```[a-zA-Z]*\s*\n/);
  if (open === null) {
    return trimmed;
  }
  return trimmed.slice(open[0].length).replace(/```\s*$/, "").trim();
}

// Real LLMs occasionally wrap the JSON in prose ("Here is the fix: {...}") —
// retry the parse on the outermost brace span.
function trySalvage(text: string): unknown {
  const first = text.indexOf("{");
  const last = text.lastIndexOf("}");
  if (first === -1 || last <= first) {
    throw new MalformedContractError("response contains no JSON object");
  }
  try {
    return JSON.parse(text.slice(first, last + 1));
  } catch {
    throw new MalformedContractError("response is not valid JSON (fenced, bare, or brace-salvaged)");
  }
}
```

// sandbox contract (script-style, sync-only, injected test/expect/assert).
export const SYSTEM_PROMPT = [
  "You are a senior TypeScript engineer implementing a small pure function.",
  "Reply with ONLY a JSON object, no prose around it, exactly:",
  '{"explanation": "<what you did>", "files": {"<filename>": "<full file content>"}}',
  "Hard rules:",
  "- The code must type-check cleanly under strict ES2022.",
  "- Every test must pass.",
  "- Return exactly ONE script-style file: NO import or export statements anywhere.",
  "- No async, no await, no Promise (the runtime is synchronous).",
  "- The runtime injects test(name, fn), expect(actual).toEqual/toBeTrue/toBeCloseTo, assert(cond, msg?) and console.",
  "- Include test(\"name\", () => { expect(...) }) calls that cover the spec.",
  "- Return the FULL content of every file every time, never a diff.",
].join("\n");

const CORRECTIVE_PROMPT = "respond with ONLY valid JSON";
const FEEDBACK_WINDOW = 3;

// CONTEXT feedback format for type errors: `file:line:col TSxxxx: message`.
function formatDiagnostics(diagnostics: TypecheckDiagnostic[]): string {
  return diagnostics
    .filter((d) => d.category === "Error")
    .map((d) => `${d.file}:${d.line}:${d.column} TS${d.code}: ${d.message}`)
    .join("\n");
}

// CONTEXT feedback for sandbox failures: failed test names + assertion
// errors + captured logs + "fix and return the FULL file(s) again".
function formatSandboxFeedback(exec: ExecuteResult): string {
  const lines: string[] = [];
  for (const failed of exec.tests.filter((t) => !t.passed)) {
    lines.push(`FAILED test "${failed.name}"${failed.error === undefined ? "" : `: ${failed.error}`}`);
  }
  for (const log of exec.logs) {
    lines.push(`guest ${log.level}: ${log.args.map((a) => JSON.stringify(a) ?? String(a)).join(" ")}`);
  }
  if (exec.status !== "ok" && exec.error !== undefined) {
    lines.push(`sandbox ${exec.status}: ${exec.error}`);
  }
  lines.push("fix and return the FULL file(s) again");
  return lines.join("\n");
}

function rootFiles(files: Record<string, string>): Map<string, string> {
  return new Map(
    Object.entries(files).map(([name, content]): [string, string] => [
      name.startsWith("/") ? name : `/${name}`,
      content,
    ]),
  );
}

export async function runAgent(deps: AgentLoopDeps): Promise<AgentRunResult> {
  const t0 = performance.now();
  const deadline = t0 + deps.runDeadlineMs;
  const iterations: AgentIteration[] = [];
  const totals = { llmMs: 0, typecheckMs: 0, transpileMs: 0, sandboxMs: 0 };
  const finish = (status: AgentStatus): AgentRunResult => ({
    status,
    iterations,
    timings: { ...totals, totalMs: performance.now() - t0 },
  });
  const baseMessages: LlmMessage[] = [
    { role: "system", content: SYSTEM_PROMPT },
    { role: "user", content: `${deps.task}\n\nSPEC:\n${deps.spec}\n\nReturn ONLY the JSON contract.` },
  ];
  const feedbackHistory: LlmMessage[] = [];

  for (let n = 1; n <= deps.maxIterations; n++) {
    // G10: the run deadline is OUR wall-clock check (workerd local dev does
    // not enforce CPU limits) — checked before every LLM call.
    if (performance.now() >= deadline) {
      return finish("run_deadline");
    }
    // CONTEXT windowing: system + task/spec first user message + last 3
    // feedback messages. No assistant-role echoes.
    const messages: LlmMessage[] = [...baseMessages, ...feedbackHistory.slice(-FEEDBACK_WINDOW)];
    let iterLlmMs = 0;

    const llmStart = performance.now();
    let text: string;
    try {
      text = (await deps.llm.complete(messages)).text;
    } catch {
      iterLlmMs = performance.now() - llmStart;
      totals.llmMs += iterLlmMs;
      return finish("llm_error");
    }
    iterLlmMs = performance.now() - llmStart;
    totals.llmMs += iterLlmMs;

    let contract: LlmContract | undefined;
    let violationFeedback: string | undefined;
    let malformedMessage: string | undefined;
    let retryLlmFailed = false;
    try {
      contract = extractContract(text);
    } catch (err) {
      if (err instanceof ScriptViolationError) {
        // G3: script violations heal via feedback, never terminally.
        violationFeedback = `${err.message}; fix and return the FULL file(s) again`;
      } else if (err instanceof MalformedContractError && performance.now() < deadline) {
        // G9: exactly one corrective retry per malformed iteration.
        const retryStart = performance.now();
        try {
          const corrective: LlmMessage = { role: "user", content: CORRECTIVE_PROMPT };
          text = (await deps.llm.complete([...messages, corrective])).text;
          contract = extractContract(text);
        } catch (retryErr) {
          if (retryErr instanceof ScriptViolationError) {
            violationFeedback = `${retryErr.message}; fix and return the FULL file(s) again`;
          } else if (retryErr instanceof MalformedContractError) {
            malformedMessage = retryErr.message;
          } else {
            retryLlmFailed = true;
          }
        } finally {
          const retryMs = performance.now() - retryStart;
          iterLlmMs += retryMs;
          totals.llmMs += retryMs;
        }
      } else if (err instanceof MalformedContractError) {
        malformedMessage = err.message;
      } else {
        malformedMessage = "contract extraction failed";
      }
    }

    if (contract === undefined && violationFeedback !== undefined) {
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: "",
        files: {},
        typecheck: null,
        transpileMs: null,
        sandbox: null,
        feedback: violationFeedback,
      });
      feedbackHistory.push({ role: "user", content: violationFeedback });
      continue;
    }
    if (contract === undefined) {
      if (retryLlmFailed) {
        const fb = "llm request failed during the corrective retry";
        iterations.push({
          n,
          llmMs: iterLlmMs,
          explanation: "",
          files: {},
          typecheck: null,
          transpileMs: null,
          sandbox: null,
          feedback: fb,
        });
        return finish("llm_error");
      }
      const fb = malformedMessage ?? "contract extraction failed";
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: "",
        files: {},
        typecheck: null,
        transpileMs: null,
        sandbox: null,
        feedback: fb,
      });
      return finish("llm_malformed");
    }

    // CORE-03 gate: a failing type-check SKIPS the sandbox entirely.
    const check = await deps.typechecker.check(rootFiles(contract.files));
    totals.typecheckMs += check.ms;
    if (!check.ok) {
      const fb = `type errors — fix and return the FULL file(s) again:\n${formatDiagnostics(check.diagnostics)}`;
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: contract.explanation,
        files: contract.files,
        typecheck: check,
        transpileMs: null,
        sandbox: null,
        feedback: fb,
      });
      feedbackHistory.push({ role: "user", content: fb });
      continue;
    }

    // Sandbox executes ONE script (REQUIREMENTS out-of-scope: multi-file
    // bundling) — enforce the system prompt's single-file rule with real
    // feedback instead of silently running an arbitrary first file.
    const names = Object.keys(contract.files);
    if (names.length !== 1) {
      const fb =
        "return exactly ONE file (the sandbox executes a single script); fix and return the FULL file(s) again";
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: contract.explanation,
        files: contract.files,
        typecheck: check,
        transpileMs: null,
        sandbox: null,
        feedback: fb,
      });
      feedbackHistory.push({ role: "user", content: fb });
      continue;
    }

    const tr = deps.transpile(contract.files[names[0]]);
    totals.transpileMs += tr.ms;
    const errorDiags = tr.diagnostics.filter((d) => d.category === "Error");
    if (errorDiags.length > 0) {
      // Near-impossible post-gate (transpileModule reports syntactic diags
      // only) — kept as a cheap tripwire with honest feedback.
      const fb = `transpile errors — fix and return the FULL file(s) again:\n${formatDiagnostics(errorDiags)}`;
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: contract.explanation,
        files: contract.files,
        typecheck: check,
        transpileMs: tr.ms,
        sandbox: null,
        feedback: fb,
      });
      feedbackHistory.push({ role: "user", content: fb });
      continue;
    }

    const exec = await deps.sandbox.execute(tr.js);
    totals.sandboxMs += exec.ms;
    // Success predicate: the guest evaluated AND registered tests AND every
    // test passed. A test-less guest is a failed iteration (guards an LLM
    // answer that "passes" by never asserting).
    if (exec.status === "ok" && exec.tests.length > 0 && exec.tests.every((t) => t.passed)) {
      iterations.push({
        n,
        llmMs: iterLlmMs,
        explanation: contract.explanation,
        files: contract.files,
        typecheck: check,
        transpileMs: tr.ms,
        sandbox: exec,
        feedback: null,
      });
      return finish("success");
    }

    const fb =
      exec.tests.length === 0
        ? 'no test() calls found; add test("name", () => { expect(...) }) calls per the spec, and return the FULL file(s) again'
        : formatSandboxFeedback(exec);
    iterations.push({
      n,
      llmMs: iterLlmMs,
      explanation: contract.explanation,
      files: contract.files,
      typecheck: check,
      transpileMs: tr.ms,
      sandbox: exec,
      feedback: fb,
    });
    feedbackHistory.push({ role: "user", content: fb });
  }

  return finish("max_iterations");
}

// carry the full transcript + timings; terminal statuses are loop.ts's
// AgentStatus composed with "running" (CONTEXT locked set — ROADMAP's
// deadline name is renamed "run_deadline" and "llm_malformed" is added; G1).
// FIFO cap at 50 runs (Map insertion-order eviction; transcripts are a few
// hundred KB worst case -> low MB, safely inside the 128 MB isolate — G8).
import type { AgentIteration, AgentStatus, AgentTimings } from "./agent/loop";

export type RunStatus = "running" | AgentStatus;

export interface RunRecord {
  id: string;
  status: RunStatus;
  task: string;
  createdAt: string;
  finishedAt: string | null;
  iterations: AgentIteration[];
  timings: AgentTimings | null;
}

export const MAX_STORED_RUNS = 50;

const runs = new Map<string, RunRecord>();

export const runRegistry = {
  create(task: string): RunRecord {
    if (runs.size >= MAX_STORED_RUNS) {
      const oldest = runs.keys().next();
      if (oldest.done !== true) {
        runs.delete(oldest.value);
      }
    }
    const run: RunRecord = {
      id: crypto.randomUUID(),
      status: "running",
      task,
      createdAt: new Date().toISOString(),
      finishedAt: null,
      iterations: [],
      timings: null,
    };
    runs.set(run.id, run);
    return run;
  },
  // Unchanged Phase 1 contract; the 404 route path rides on this.
  get(id: string): RunRecord | undefined {
    return runs.get(id);
  },
  // Mutates the stored record in place — the caller's `create` reference IS
  // the stored record, so the POST handler serializes it after finalize.
  // finishedAt uses wall-clock ISO time (transcript timing fidelity is
  // performance.now()'s job; this is registry bookkeeping).
  finalize(id: string, status: RunStatus, iterations: AgentIteration[], timings: AgentTimings | null): void {
    const run = runs.get(id);
    if (run === undefined) {
      return;
    }
    run.status = status;
    run.iterations = iterations;
    run.timings = timings;
    run.finishedAt = new Date().toISOString();
  },
};
