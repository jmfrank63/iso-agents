# Plan 02-01: Core checker — Text-rule libs, Map VFS + Design-A host, Typechecker seam, route wiring

<objective>
Build the Phase 2 core: real TypeScript diagnostics produced entirely inside the workerd
isolate with zero host filesystem access. Bundle the 57-file `lib.es2022.d.ts` reference
closure as wrangler Text modules (TYPE-02), implement the hand-rolled Map-based CompilerHost
(Design A: no `lib` compiler option, bare-name keys, memoized lib SourceFiles, fresh guest
Program per check), expose it behind the `Typechecker` interface with a pinned
`typescript@6.0.3` boot assert (TYPE-01) and a selectable transpile-only second
implementation proving the seam (TYPE-04), and wire `POST /typecheck` to a real one-shot
check (TYPE-03's N-run stats stay Phase 4). healthz `typechecker` flips from "unloaded" to
`typescript@6.0.3 (in-isolate)` after first use via the Phase 1 `getRuntimeInfo()` seam.
</objective>

## Locked decisions this plan implements (from 02-CONTEXT.md + 02-RESEARCH.md)

- **Pin**: `typescript@6.0.3` exact (already in package.json — do not touch it); boot assert
  `ts.version === "6.0.3"` + `createProgram`/`getPreEmitDiagnostics` typeof + `process`
  must be undefined. Fails loudly on accidental native-7.x upgrade.
- **Lib delivery**: wrangler Text rule (`**/*.d.ts`, `"fallthrough": true` — REQUIRED or the
  default `**/*.txt|html|sql` rule is silently dropped). 57 imports generated by a
  committed script, run once, output committed.
- **Host**: fully hand-rolled Map CompilerHost — never `ts.sys`/`ts.createCompilerHost`.
  **NEVER set the `lib` compiler option** (reroutes lib acquisition through
  `pathForLibFile()`, silently serves zero libs, TS2318/TS2583 storm — research gotcha #1).
  `getDefaultLibFileName(): "lib.es2022.d.ts"` is passed verbatim to `getSourceFile` and
  MUST be a map key. Lib+globals SourceFiles memoized; guest SourceFiles fresh per check.
- **Seam**: `Typechecker { id; check(files: Map<string,string>): Promise<TypecheckResult> }`;
  result `{ok, diagnostics: [{code, message, line, column, file, category}], ms}`;
  diagnostics 1-based, sorted by (file, line, column); `ok:false` iff any Error-category.
- **Init**: lazy module-level singleton; identity flip observable on healthz.
- **Timing**: `performance.now()` delta around program+getPreEmitDiagnostics inside request
  scope (workerd freezes `Date.now()` in global scope — never init there).
- **Route**: `POST /typecheck` flips 501 → real check of `{files:{name:content}}`, envelope
  `{ok:true,data:{ok,diagnostics,ms}}`, 200 even when `data.ok === false`. One run per
  request (N-run stats are TYPE-03, Phase 4).

```js
// Generates src/typecheck/lib-imports.ts: the complete lib.es2022.d.ts
// reference closure from node_modules/typescript/lib as wrangler Text-rule
// imports (default export = file text). Run once and commit; regenerate only
// when the typescript pin changes — the boot assert in src/typecheck/index.ts
// catches drift.
import { readFileSync, writeFileSync } from "node:fs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";

const root = join(dirname(fileURLToPath(import.meta.url)), "..");
const libDir = join(root, "node_modules", "typescript", "lib");
const outFile = join(root, "src", "typecheck", "lib-imports.ts");
const HEAD = "lib.es2022.d.ts";

function refsOf(fileName) {
  const text = readFileSync(join(libDir, fileName), "utf8");
  return [...text.matchAll(/^\/\/\/\s*<reference\s+lib="([^"]+)"\s*\/>\s*$/gm)].map(
    (m) => `lib.${m[1]}.d.ts`,
  );
}

const closure = new Set([HEAD]);
for (const queue = [HEAD]; queue.length > 0; ) {
  const f = queue.shift();
  for (const ref of refsOf(f)) {
    if (!closure.has(ref)) {
      closure.add(ref);
      queue.push(ref);
    }
  }
}
const files = [...closure].sort();

function ident(fileName) {
  return (
    "lib" +
    fileName
      .replace(/^lib\./, "")
      .replace(/\.d\.ts$/, "")
      .split(".")
      .map((s) => s[0].toUpperCase() + s.slice(1))
      .join("")
  );
}

const lines = [
  "// @ts-nocheck — .d.ts files are imported as wrangler Text modules (strings);",
  "// host tsc rejects declaration-file imports (TS2306/TS2846) and an ambient",
  "// wildcard shim is shadowed by real-file resolution, so this generated file",
  "// opts out of checking. The Record<string, string> annotation below keeps",
  "// consumers typed.",
  "// Generated by scripts/gen-lib-imports.mjs — do not edit by hand.",
  `// ${files.length} files = the lib.es2022.d.ts reference closure (no DOM).`,
];
for (const f of files) {
  lines.push(`import ${ident(f)} from "typescript/lib/${f}";`);
}
lines.push("");
lines.push("export const LIB_FILES: Record<string, string> = {");
for (const f of files) {
  lines.push(`  "${f}": ${ident(f)},`);
}
lines.push("};");
lines.push("");
writeFileSync(outFile, lines.join("\n"));
console.log(`wrote ${outFile}: ${files.length} imports`);
```

1. **src/typecheck/vfs.ts** (new file, exact content):

```ts
import ts from "typescript";
import { LIB_FILES } from "./lib-imports";

// Bare basenames are the map keys: with no `lib` compiler option, typescript
// hands getDefaultLibFileName()'s return value verbatim to getSourceFile(),
// and /// <reference lib> chains also use bare names. Any other key shape
// (absolute paths, "es2022") silently misses the map.
export const LIB_TEXT: ReadonlyMap<string, string> = new Map(Object.entries(LIB_FILES));

// The es2022 closure declares no `console`; guests are contractually allowed
// console.log/error (Phase 3 shims them). Declaration files load only when
// reachable, so this file must stay in every check's rootNames.
export const GUEST_GLOBALS = "/guest-globals.d.ts";
export const GUEST_GLOBALS_TEXT =
  "declare var console: { log(...args: unknown[]): void; error(...args: unknown[]): void; };";

// Lib + globals SourceFiles are immutable pinned text: parse once per
// isolate, reuse across checks (research: 223 ms cold -> 8-20 ms warm). Guest
// SourceFiles are parsed fresh every check — their text changes per
// iteration. Cache holds 58 entries at steady state (57 libs + globals).
const sourceFileCache = new Map<string, ts.SourceFile>();

export function memoSourceFile(
  fileName: string,
  text: string,
  languageVersionOrOptions: ts.ScriptTarget | ts.CreateSourceFileOptions,
): ts.SourceFile {
  let sf = sourceFileCache.get(fileName);
  if (sf === undefined) {
    sf = ts.createSourceFile(fileName, text, languageVersionOrOptions, true);
    sourceFileCache.set(fileName, sf);
  }
  return sf;
}
```

2. **src/typecheck/host.ts** (new file, exact content):

```ts
import ts from "typescript";
import { GUEST_GLOBALS, GUEST_GLOBALS_TEXT, LIB_TEXT, memoSourceFile } from "./vfs";

// Guest-check compiler options. NEVER add a `lib` option: with `lib` set,
// typescript routes lib acquisition through pathForLibFile() (raw option
// string + directory prefix) and never calls getDefaultLibFileName(), so a
// bare-name VFS map serves zero libs and every check drowns in TS2318/
// TS2583. Verified in typescript@6.0.3 (ts.js L127725-L127749).
export const GUEST_COMPILER_OPTIONS: ts.CompilerOptions = {
  strict: true,
  target: ts.ScriptTarget.ES2022,
  noEmit: true,
  module: ts.ModuleKind.ESNext,
  moduleResolution: ts.ModuleResolutionKind.Bundler,
  skipLibCheck: true,
  isolatedModules: true,
  forceConsistentCasingInFileNames: true,
};

// Design A host (research-verified in workerd): every file — libs included —
// is served from maps; nothing touches a host filesystem. Lib + globals
// SourceFiles are memoized; guest files re-parse per check. Guest paths are
// rooted ("/guest.ts"); lib keys are bare ("lib.es2022.d.ts").
export function makeHost(guestFiles: Map<string, string>): ts.CompilerHost {
  return {
    getSourceFile(fileName, languageVersionOrOptions) {
      if (fileName === GUEST_GLOBALS) {
        return memoSourceFile(fileName, GUEST_GLOBALS_TEXT, languageVersionOrOptions);
      }
      const libText = LIB_TEXT.get(fileName);
      if (libText !== undefined) {
        return memoSourceFile(fileName, libText, languageVersionOrOptions);
      }
      const text = guestFiles.get(fileName);
      return text === undefined
        ? undefined
        : ts.createSourceFile(fileName, text, languageVersionOrOptions, true);
    },
    getDefaultLibFileName: () => "lib.es2022.d.ts", // MUST be a LIB_TEXT key
    writeFile: () => {}, // noEmit: nothing to write, ever
    getCurrentDirectory: () => "/",
    getDirectories: () => [],
    fileExists: (f) => f === GUEST_GLOBALS || LIB_TEXT.has(f) || guestFiles.has(f),
    readFile: (f) => LIB_TEXT.get(f) ?? guestFiles.get(f),
    readDirectory: () => [],
    getCanonicalFileName: (f) => f,
    useCaseSensitiveFileNames: () => true,
    getNewLine: () => "\n",
  };
}
```

**src/typecheck/index.ts** (new file, exact content):

```ts
import ts from "typescript";
import { GUEST_COMPILER_OPTIONS, makeHost } from "./host";
import { GUEST_GLOBALS } from "./vfs";

export interface TypecheckDiagnostic {
  code: number;
  message: string;
  /** 1-based; -1 when the diagnostic has no file/start (global/config diags). */
  line: number;
  /** 1-based; -1 when the diagnostic has no file/start. */
  column: number;
  file: string;
  category: string;
}

export interface TypecheckResult {
  ok: boolean;
  diagnostics: TypecheckDiagnostic[];
  ms: number;
}

// tsc-rs drop-in seam. tsc-rs ships native binaries only today (no WASM
// build), so it cannot run inside workerd; this interface is the documented
// boundary a future WASM tsc-rs build implements. The agent loop (Phase 4)
// and the /typecheck benchmark depend on `Typechecker`, never on tsc
// directly — swapping implementations changes nothing above this line.
export interface Typechecker {
  readonly id: string;
  check(files: Map<string, string>): Promise<TypecheckResult>;
}

// Boot-time identity assert (TYPE-01): an accidental upgrade to npm "latest"
// (native tsgo 7.x) must fail loudly here, not silently break in-isolate
// checking. Runs once, on the first getTypechecker() call.
function assertCompilerIdentity(): void {
  if (ts.version !== "6.0.3") {
    throw new Error(
      `FATAL: in-isolate typechecker requires exactly typescript@6.0.3 (pure-JS line); ` +
        `loaded "${ts.version}". npm "latest" (7.x) is the NATIVE tsgo build and cannot run in workerd.`,
    );
  }
  if (typeof ts.createProgram !== "function" || typeof ts.getPreEmitDiagnostics !== "function") {
    throw new Error("FATAL: typescript createProgram/getPreEmitDiagnostics missing — API shape changed");
  }
  if (typeof (globalThis as Record<string, unknown>).process !== "undefined") {
    throw new Error(
      "FATAL: process global present — no_nodejs_compat flags lost; isNodeLikeSystem() would take the Node branch",
    );
  }
}

function mapDiagnostic(d: ts.Diagnostic): TypecheckDiagnostic {
  const loc =
    d.file !== undefined && d.start !== undefined
      ? ts.getLineAndCharacterOfPosition(d.file, d.start) // 0-based
      : { line: -1, character: -1 }; // global/config diags carry no file/start
  return {
    code: d.code,
    message: ts.flattenDiagnosticMessageText(d.messageText, "\n"),
    line: loc.line + 1,
    column: loc.character + 1,
    file: d.file?.fileName ?? "(program)",
    category: ts.DiagnosticCategory[d.category],
  };
}

function byLocation(a: TypecheckDiagnostic, b: TypecheckDiagnostic): number {
  return a.file.localeCompare(b.file) || a.line - b.line || a.column - b.column;
}

export class TsTypechecker implements Typechecker {
  readonly id = "typescript@6.0.3 (in-isolate)";

  async check(files: Map<string, string>): Promise<TypecheckResult> {
    const host = makeHost(files);
    const t0 = performance.now();
    const program = ts.createProgram({
      rootNames: [GUEST_GLOBALS, ...files.keys()],
      options: GUEST_COMPILER_OPTIONS,
      host,
    });
    const diagnostics = ts.getPreEmitDiagnostics(program);
    const ms = performance.now() - t0;
    const mapped = diagnostics.map(mapDiagnostic).sort(byLocation);
    return { ok: mapped.every((d) => d.category !== "Error"), diagnostics: mapped, ms };
  }
}

// Second implementation (TYPE-04 seam proof): same interface, no
// createProgram — transpiles each file, always reports ok. A future tsc-rs
// WASM build plugs in exactly like this class does.
export class TranspileOnlyTypechecker implements Typechecker {
  readonly id = "transpile-only (seam proof)";

  async check(files: Map<string, string>): Promise<TypecheckResult> {
    const t0 = performance.now();
    for (const text of files.values()) {
      ts.transpileModule(text, {
        compilerOptions: { target: ts.ScriptTarget.ES2022, isolatedModules: true },
      });
    }
    return { ok: true, diagnostics: [], ms: performance.now() - t0 };
  }
}

export type TypecheckerKind = "ts" | "transpile-only";

export function createTypechecker(kind: TypecheckerKind = "ts"): Typechecker {
  return kind === "ts" ? new TsTypechecker() : new TranspileOnlyTypechecker();
}

// Lazy module-level singleton (CONTEXT decision): boot assert + construction
// on first call; lib parse happens inside the first real check.
// getTypecheckerIdentity() never triggers init, so /healthz stays fast and
// flips only after the checker is genuinely used.
let singleton: Typechecker | undefined;

export function getTypechecker(): Typechecker {
  if (singleton === undefined) {
    assertCompilerIdentity();
    singleton = new TsTypechecker();
  }
  return singleton;
}

export function getTypecheckerIdentity(): string {
  return singleton !== undefined ? `typescript@${ts.version} (in-isolate)` : "unloaded";
}
```

```ts
interface TypecheckBody {
  files?: Record<string, unknown>;
}

// TYPE-02 early HTTP path: one wall-ms check per request (TYPE-03's N-run
// stats land in Phase 4). Snippet names are VFS-rooted: "guest.ts" is
// checked as "/guest.ts". Responds 200 with data.ok=false when the checked
// code has errors — the check itself succeeded.
async function handleTypecheck(request: Request): Promise<Response> {
  const body = (await parseJsonBody(request)) as TypecheckBody | null;
  if (
    body === null ||
    typeof body !== "object" ||
    body.files === undefined ||
    body.files === null ||
    typeof body.files !== "object"
  ) {
    throw new HttpError(400, "bad_request", "body must be JSON {files: {name: content}}");
  }
  const entries = Object.entries(body.files);
  if (entries.length === 0) {
    throw new HttpError(400, "bad_request", "files must contain at least one entry");
  }
  const bad = entries.find(([, content]) => typeof content !== "string");
  if (bad !== undefined) {
    throw new HttpError(400, "bad_request", `files.${bad[0]} must be a string`);
  }
  const files = new Map<string, string>(
    entries.map(([name, content]): [string, string] => [
      name.startsWith("/") ? name : `/${name}`,
      content as string,
    ]),
  );
  const result = await getTypechecker().check(files);
  return json({ ok: true, data: result }, 200);
}
```
